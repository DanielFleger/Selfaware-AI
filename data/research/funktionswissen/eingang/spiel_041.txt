 DAT_BuildingsState.buildings[DAT_CurrentBuildingID].animationCycleCount += 1;
 DAT_BuildingsState.buildings[DAT_CurrentBuildingID].animationCycleCompleted = 1;
 DAT_BuildingsState.buildings[DAT_CurrentBuildingID].state = 0;
 }
 }
 if (DAT_BuildingsState.buildings[DAT_CurrentBuildingID].renderBlendStrength > 31) {
 DAT_BuildingsState.buildings[DAT_CurrentBuildingID].renderBlendStrength = 31;
 } else if (DAT_BuildingsState.buildings[DAT_CurrentBuildingID].renderBlendStrength < 0) {
 DAT_BuildingsState.buildings[DAT_CurrentBuildingID].renderBlendStrength = 0;
 }
 int const workerID = DAT_BuildingsState.buildings[DAT_CurrentBuildingID].workerID[0];
 if (workerID != 0) {
 if (DAT_UnitsState.units[workerID].state.generic
 == US_AIM_WEAPONUnk) {
 DAT_BuildingsState.buildings[DAT_CurrentBuildingID].buildingIsVisuallyActive = 1;
 } else if (DAT_UnitsState.units[workerID].state.generic
 == US_RELOAD_WEAPONUnk) {
 DAT_BuildingsState.buildings[DAT_CurrentBuildingID].buildingIsVisuallyActive = 1;
 } else {
 DAT_BuildingsState.buildings[DAT_CurrentBuildingID].buildingIsVisuallyActive = 0;
 }
 } else {
 DAT_BuildingsState.buildings[DAT_CurrentBuildingID].buildingIsVisuallyActive = 0;
 }
 updateVisuallyActiveState(DAT_CurrentBuildingID);
 if (DAT_BuildingsState.buildings[DAT_CurrentBuildingID].buildingIsVisuallyActive
 != DAT_BuildingsState.buildings[DAT_CurrentBuildingID].oldVisualActiveState) {
 updateBuildingGraphicsLayer(
 DAT_CurrentBuildingID);
 DAT_BuildingsState.buildings[DAT_CurrentBuildingID].oldVisualActiveState
 = DAT_BuildingsState.buildings[DAT_CurrentBuildingID].buildingIsVisuallyActive;
 }
 if (DAT_GameSynchronyState.currentGameMode != 0) {
 DAT_BuildingsState.buildings[DAT_CurrentBuildingID].displayOwnerFlag = 1;
 DAT_BuildingsState.buildings[DAT_CurrentBuildingID].flagSlot.overlayImage += 1;
 if (DAT_BuildingDefinedData.SharedOverlayAnimationFrames
 [DAT_BuildingsState.buildings[DAT_CurrentBuildingID].flagSlot.overlayImage / 2]
 <= 0) {
 DAT_BuildingsState.buildings[DAT_CurrentBuildingID].flagSlot.overlayImage = 0;
 }
 DAT_BuildingsState.buildings[DAT_CurrentBuildingID].overlayImageID
 = DAT_BuildingDefinedData.SharedOverlayAnimationFrames
 [DAT_BuildingsState.buildings[DAT_CurrentBuildingID].flagSlot.overlayImage
 / 2];
 } else {
 DAT_BuildingsState.buildings[DAT_CurrentBuildingID].overlayImageID = 0;
 }
 }
}
