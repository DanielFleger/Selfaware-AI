 _ucrt_write(_fileDescriptor, this->tempMem, this->directory.filePayloadSize);
 _ucrt_close(_fileDescriptor);
 _free_base(this->tempMem);
 _free_base(_tempMem2);
 this->tempMem = NULL;
 }
