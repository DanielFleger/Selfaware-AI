 int Units::ComputeGoodsProduced(int unitID, int goodsCount, BOOLEnum boost)
 {
 short _productionBonusGoods;
 int _productivity;
 int _productionBonusPercentage;
 _productivity = DAT_GameState.playerDataArray[DAT_UnitsState.units[unitID].owner]
 .fearFactorProductivityUnk;
 DAT_UnitsState.units[unitID].productivityDiv100 = 0;
 /*
 if productivity less than 101, set it to 100. I guess because decreased productivity level (80%) means units
 take more breaks, not produce less than 100% level.
 */
 if (_productivity < 100) {
 _productivity = 100;
 }
 if ((DAT_GameSynchronyState.currentGameMode != GM_SOLITARY) && (boost != FALSE)) {
 /*
 boost productivity by +50 in skirmish modes
 */
 _productivity = _productivity + 50;
 }
 _productionBonusPercentage = ((_productivity + -100) * goodsCount * 100) / 100
 + (int)DAT_UnitsState.units[unitID].productivityMod100;
 _productionBonusGoods = (short)(_productionBonusPercentage / 100);
 DAT_UnitsState.units[unitID].productivityDiv100 = _productionBonusGoods;
 /*
 for example, if production was 150%, then 50% extra goods were produced, but 0.5 goods do not exist, so this
 bonus is stored in this variable (in percentage form: 50%), and added next time around. Assuming goodsCount
 = 1, productivity = 150, this produces the production pattern: [1, 2, 1, 2, 1, 2]
 */
 DAT_UnitsState.units[unitID].productivityMod100 = (short)(_productionBonusPercentage % 100);
 return _productionBonusGoods + goodsCount;
