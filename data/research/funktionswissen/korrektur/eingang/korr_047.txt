 {
 int const aivID = DAT_GameState.playerDataArray[playerID].aivID;
 int x;
 int y;
 MappersEnum const mapper = (MappersEnum)this->aivs[aivID].aivBuildingSteps[step].buildingType;
 // when pausing, only hovels are built

 return FALSE;

 // disabled steps and steps without room are never built
 if (this->aivs[aivID].aivBuildingSteps[step].buildStatus == AIVBSS_disabled) {
 return FALSE;

 if (this->aivs[aivID].aivBuildingSteps[step].buildStatus == AIVBSS_insufficient_roomUnk) {

 || mapper == Commands::M_MAPPER_STAIR5 || mapper == Commands::M_MAPPER_STAIR6) {

 orientation = 80;

 if (this->aivs[aivID].keepOrientation == 2 || this->aivs[aivID].keepOrientation == 6) {
 orientation = 81;
 } else {
 orientation = 80;
 }
 } else if (mapper == Commands::M_MAPPER_POLETURNER || mapper == Commands::M_MAPPER_BLACKSMITH
 || mapper == Commands::M_MAPPER_ARMOURER || mapper == Commands::M_MAPPER_TANNER
 || mapper == Commands::M_MAPPER_FLETCHER || mapper == Commands::M_MAPPER_BAKER
 || mapper == Commands::M_MAPPER_BREWER) {
 orientation = 8;
 } else if (mapper == Commands::M_MAPPER_DRAWBRIDGE) {
 DAT_TileMapState.buildingPlacementFail = FALSE;
 checkDrawbridgePlacement(x, y);
 if (DAT_TileMapState.buildingPlacementFail) {
 return FALSE;
 }
 orientation = DAT_TileMapState.uiBuildingRotation;
 }
 if (this->aivs[aivID].aivBuildingSteps[step].buildStatus == AIVBSS_built) {
 // rebuild a destroyed building
 int const buildingID = (short)DAT_TileMapState.BuildingLayer[tile];
 if ((short)DAT_BuildingsState.buildings[buildingID].buildingType
 == convertCommandBuildingTypeToBuildingType(mapper)) {
 return FALSE;
 }
 if (DAT_GameState.playerDataArray[playerID].aiBuildingDestroyChoiceTracker == 0
 && DAT_GameState.playerDataArray[playerID].aiNervousActionsTracker == 0
 && this->aivs[aivID].aivBuildingSteps[step].wait > 0) {
 --this->aivs[aivID].aivBuildingSteps[step].wait;
 return FALSE;
 }
 if (!checkRequiredResourcesForBuildingOrPlanToBuy(mapper, playerID, FALSE)) {
 return FALSE;
 }
 }
 if (shouldWaitWithBuildingAPositiveFearFactor(
 playerID, mapper)) {
 return FALSE;
 }
 if (shouldWaitWithBuildingNegativeFearFactor(
 playerID, mapper)) {
 return FALSE;
 }
 if (orientation == 15) {

 } else {
 DAT_TileMapState.DAT_TempBuildingRotation = orientation / 2;
 }
 int const buildingSize = getBuildingSizeForCommandBuildingType(mapper);
 prepareAreaForBuildingPlacement(
 playerID, x, y, mapper, buildingSize);
 // keep orientation is 0, 2, 4 or 6, halving it gives the index of the part offsets
 if (mapper == Commands::M_MAPPER_BARRACKS_EURO || mapper == Commands::M_MAPPER_BARRACKS_ARAB) {

 prepareAreaForBuildingPlacement(
 playerID,
 DAT_TerrainDefinedData.BuildingPartsOffsets[this->aivs[aivID].keepOrientation / 2][0].x
 + x,
 DAT_TerrainDefinedData.BuildingPartsOffsets[this->aivs[aivID].keepOrientation / 2][0].y
 + y,
 mapper, buildingSize);
 prepareAreaForBuildingPlacement(
 playerID,
 DAT_TerrainDefinedData.BuildingPartsOffsets[this->aivs[aivID].keepOrientation / 2][1].x
 + x,
 DAT_TerrainDefinedData.BuildingPartsOffsets[this->aivs[aivID].keepOrientation / 2][1].y
 + y,
 mapper, buildingSize);
 prepareAreaForBuildingPlacement(
 playerID,
 DAT_TerrainDefinedData.BuildingPartsOffsets[this->aivs[aivID].keepOrientation / 2][2].x
 + x,
 DAT_TerrainDefinedData.BuildingPartsOffsets[this->aivs[aivID].keepOrientation / 2][2].y
 + y,
 mapper, buildingSize);
 } else if (mapper == Commands::M_MAPPER_ENGINEERS_GUILD || mapper == Commands::M_MAPPER_TUNNELERS_GUILD) {
 // the guild field
 prepareAreaForBuildingPlacement(
 playerID,
 DAT_TerrainDefinedData.BuildingPartsOffsets[this->aivs[aivID].keepOrientation / 2][1].x
 + x,
 DAT_TerrainDefinedData.BuildingPartsOffsets[this->aivs[aivID].keepOrientation / 2][1].y
 + y,
 mapper, buildingSize);
 } else if (mapper == Commands::M_MAPPER_OIL_SMELTER) {
 prepareAreaForBuildingPlacement(
 playerID,
 DAT_TerrainDefinedData.field63_0x19c[this->aivs[aivID].keepOrientation / 2].x + x,
 DAT_TerrainDefinedData.field63_0x19c[this->aivs[aivID].keepOrientation / 2].y + y, mapper,
 buildingSize);
 }
 placeBuilding(playerID, x, y, mapper,
 getBuildingSizeForCommandBuildingType(
 mapper),
 orientation);
 int const placedID = DAT_TileMapState.placedBuildingID;
 if (!DAT_TileMapState.buildingPlacementFail) {
 DAT_BuildingsState.buildings[placedID].field201_0x2c8 = 1;
 setupWorkshopProductionType(playerID, placedID);
 }
 if (!DAT_TileMapState.buildingPlacementFail) {
 this->aivs[aivID].aivBuildingSteps[step].buildStatus = AIVBSS_built;
 } else if (mapper == Commands::M_MAPPER_WHEATFARM || mapper == Commands::M_MAPPER_HOPSFARM
 || mapper == Commands::M_MAPPER_APPLEFARM || mapper == Commands::M_MAPPER_CATTLEFARM) {
 this->aivs[aivID].aivBuildingSteps[step].buildStatus = AIVBSS_insufficient_roomUnk;
 } else {
 this->aivs[aivID].aivBuildingSteps[step].buildStatus = AIVBSS_insufficient_resources;
 }
 this->aivs[aivID].aivBuildingSteps[step].wait
 = this->aivs[aivID].aivBuildingSteps[step].buildStatus == AIVBSS_built ? 10 : 16;
 if (!DAT_TileMapState.buildingPlacementFail) {
 return TRUE;
 }
 }
 return FALSE;
 }
}

 void AIVState::findAppropriateGridLocationForResourceTypeBuilding(
 int playerID, AIGridComputationResourceType resourceType)
 {
 this->visitCount = 1;
 this->algAIndex = 0;
 this->algBIndex = 1;
 this->buildingApproriateGridXPosition = -1;
 this->buildingAppropriateGridYPosition = -1;
 if (resourceType == AIGCRT_OIL) {
 if (DAT_GameState.playerDataArray[playerID].countPitchRigs
 >= DAT_GameState.playerDataArray[playerID].availableOilGridTiles) {
 return;
 }
 if (DAT_GameState.playerDataArray[playerID].algoOilCooldown > 0) {
 --DAT_GameState.playerDataArray[playerID].algoOilCooldown;
 return;
 }
 } else if (resourceType == AIGCRT_IRON) {
 if (DAT_GameState.playerDataArray[playerID].countIronMines
 >= DAT_GameState.playerDataArray[playerID].availableIronGridTiles) {
 return;
 }
 if (DAT_GameState.playerDataArray[playerID].algoIronCooldown > 0) {
 --DAT_GameState.playerDataArray[playerID].algoIronCooldown;
 return;
 }
 } else if (resourceType == AIGCRT_BOULDER) {
 if (DAT_GameState.playerDataArray[playerID].countStoneQuarries
 >= DAT_GameState.playerDataArray[playerID].availableBoulderGridTiles) {
 return;
 }
 if (DAT_GameState.playerDataArray[playerID].algoBoulderCooldown > 0) {
 --DAT_GameState.playerDataArray[playerID].algoBoulderCooldown;
 return;
 }
 }
 int const campX = DAT_GameState.playerDataArray[playerID].campground.xEntry / 5;
 this->heatMapXArray[0] = campX;
 int const campY = DAT_GameState.playerDataArray[playerID].campground.yEntry / 5;
 ++this->mapExtraInfo.algorithmIterationNumber;
 this->heatMapYArray[0] = campY;
 this->heatMaps[campX][campY].algorithmIterationNumber = this->mapExtraInfo.algorithmIterationNumber;
 this->heatMaps[campX][campY].algorithmVisitCountUnk = this->visitCount;
 while (this->algAIndex != this->algBIndex) {
 int const x = this->heatMapXArray[this->algAIndex];
 int const y = this->heatMapYArray[this->algAIndex];
 this->visitCount = this->heatMaps[x][y].algorithmVisitCountUnk;
 if (this->visitCount > 40) {
 break;
 }
 // only the four cardinal directions
 for (int direction = 0; direction < 8; direction += 2) {
 uint const gridX
 = DAT_TerrainDefinedData.clockwiseCardinalTranslationMatrix[direction].int_.xOffset + x;
 uint const gridY
 = DAT_TerrainDefinedData.clockwiseCardinalTranslationMatrix[direction].int_.yOffset + y;
 if (this->heatMaps[x][y].isNotNearEdgeUnk && (gridX > 79 || gridY > 79)) {
 continue;
 }
 if (this->heatMaps[gridX][gridY].algorithmIterationNumber
 == this->mapExtraInfo.algorithmIterationNumber) {
 continue;
 }
 this->heatMaps[gridX][gridY].algorithmIterationNumber = this->mapExtraInfo.algorithmIterationNumber;
 int const blockedCount = this->heatMaps[gridX][gridY].tilesNotPartOfLargestAreaCount
 - this->heatMaps[gridX][gridY].impassableCount;
 if (blockedCount >= 16) {
 continue;

 int const owner = this->heatMaps[gridX][gridY].tileOwnership;
 if ((blockedCount == 0 || resourceType == AIGCRT_IRON) && blockedCount <= 4
 && !this->heatMaps[gridX][gridY].structureCount
 && !this->heatMaps[gridX][gridY].destructionBasedPlacementCooldown
 && (!this->heatMaps[gridX][gridY].tileOwnership
 || DAT_GameState.mapAndTime.playerTeams[playerID]
 == DAT_GameState.mapAndTime.playerTeams[owner])) {
 if (resourceType == AIGCRT_OIL) {
 if (this->heatMaps[gridX][gridY].oilCount > 2 && this->heatMaps[gridX][gridY].marshCount >= 10
 && this->heatMaps[gridX][gridY].highestTerrainHeight
 - this->heatMaps[gridX][gridY].lowestTerrainHeight
 < 12) {
 this->buildingApproriateGridXPosition = gridX;
 this->buildingAppropriateGridYPosition = gridY;
 return;
 }
 } else if (resourceType == AIGCRT_IRON) {
 if (this->heatMaps[gridX][gridY].ironCount > 6
 && this->heatMaps[gridX][gridY].highestTerrainHeight
 - this->heatMaps[gridX][gridY].lowestTerrainHeight
 < 12) {
 this->buildingApproriateGridXPosition = gridX;
 this->buildingAppropriateGridYPosition = gridY;
 return;
 }
 } else if (resourceType == AIGCRT_BOULDER) {
 if (this->heatMaps[gridX][gridY].bouldersCount >= 8
 && this->heatMaps[gridX][gridY].highestTerrainHeight
 - this->heatMaps[gridX][gridY].lowestTerrainHeight
 < 40) {
 this->buildingApproriateGridXPosition = gridX;
 this->buildingAppropriateGridYPosition = gridY;
 return;
 }
 }
 }
 this->heatMaps[gridX][gridY].algorithmVisitCountUnk = this->visitCount + 1;
 this->heatMapXArray[this->algBIndex] = gridX;
 this->heatMapYArray[this->algBIndex] = gridY;
 ++this->algBIndex;
 if (this->algBIndex >= 6400) {
 this->algBIndex = 0;
 }
 }
 ++this->algAIndex;
 if (this->algAIndex >= 6400) {
 this->algAIndex = 0;
 }
 }
 // nothing found, wait before searching again
 if (resourceType == AIGCRT_OIL) {
 DAT_GameState.playerDataArray[playerID].algoOilCooldown = 5;
 return;
 }
 if (resourceType == AIGCRT_IRON) {
 DAT_GameState.playerDataArray[playerID].algoIronCooldown = 5;
 return;
 }
 if (resourceType == AIGCRT_BOULDER) {
 DAT_GameState.playerDataArray[playerID].algoBoulderCooldown = 5;
 }
 }
}
